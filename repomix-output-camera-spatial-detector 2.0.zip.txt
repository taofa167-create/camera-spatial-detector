File: src/lib/spatial.ts
================
 1: export type BBox = [number, number, number, number];
 2: 
 3: export type Relation = "on top of" | "below" | "beside";
 4: 
 5: /**
 6:  * Topological spatial engine.
 7:  * Compares two bounding boxes and decides whether object 1 is
 8:  * "on top of", "below", or "beside" object 2, using vertical and
 9:  * horizontal midpoint coordinates.
10:  */
11: export function resolveRelation(b1: BBox, b2: BBox): Relation {
12:   const midY1 = b1[1] + b1[3] / 2;
13:   const midY2 = b2[1] + b2[3] / 2;
14:   const midX1 = b1[0] + b1[2] / 2;
15:   const midX2 = b2[0] + b2[2] / 2;
16: 
17:   const verticalGap = Math.abs(midY1 - midY2);
18:   const horizontalGap = Math.abs(midX1 - midX2);
19: 
20:   // Clear vertical stacking: one midpoint sits above the other box's edge.
21:   if (midY1 < b2[1]) return "on top of";
22:   if (midY2 < b1[1]) return "below";
23: 
24:   // Otherwise fall back to the dominant axis between the midpoints.
25:   if (verticalGap > horizontalGap) {
26:     return midY1 < midY2 ? "on top of" : "below";
27:   }
28:   return "beside";
29: }
30: 
31: export function describeRelation(
32:   class1: string,
33:   bbox1: BBox,
34:   class2: string,
35:   bbox2: BBox,
36: ): string {
37:   return `The ${class1} is ${resolveRelation(bbox1, bbox2)} the ${class2}`;
38: }

 1: # Routes
 2: 
 3: TanStack Start uses **file-based routing**. Every `.tsx` file in this directory
 4: defines a route. Do **not** create `src/pages/`, `src/routes/_app/index.tsx`, or
 5: `app/layout.tsx` — those are Next.js / Remix conventions. The only root layout
 6: is `src/routes/__root.tsx`.
 7: 
 8: ## Conventions
 9: 
10: | File | URL |
11: | --- | --- |
12: | `index.tsx` | `/` |
13: | `about.tsx` | `/about` |
14: | `users/index.tsx` | `/users` |
15: | `users/$id.tsx` | `/users/:id` (dynamic — bare `$`, no curly braces) |
16: | `posts/{-$category}.tsx` | `/posts/:category?` (optional segment) |
17: | `files/$.tsx` | `/files/*` (splat — read via `_splat` param, never `*`) |
18: | `_layout.tsx` | layout route (renders children via `<Outlet />`) |
19: | `__root.tsx` | app shell — wraps every page; preserve `<Outlet />` |
20: 
21: `routeTree.gen.ts` is auto-generated. Don't edit it by hand.

16: // h3 swallows in-handler throws into a normal 500 Response with body
17: // {"unhandled":true,"message":"HTTPError"} — try/catch alone never fires for those.
18: async function normalizeCatastrophicSsrResponse(response: Response): Promise<Response> {
19:   if (response.status < 500) return response;
20:   const contentType = response.headers.get("content-type") ?? "";
21:   if (!contentType.includes("application/json")) return response;
22: 
23:   const body = await response.clone().text();
24:   if (!isH3SwallowedErrorBody(body)) return response;
25: 
26:   console.error(body);
27: return new Response("<h1>500 Error</h1>", {
28:     status: 500,
29:     headers: { "content-type": "text/html; charset=utf-8" },
30:   });
31: }
32: 
33: function isH3SwallowedErrorBody(body: string): boolean {
34:   try {
35:     const payload = JSON.parse(body) as { unhandled?: unknown; message?: unknown };
36:     return payload.unhandled === true && payload.message === "HTTPError";
37:   } catch {
38:     return false;
39:   }
40: }
41:
